(function () {
            var overlay = document.getElementById('csrLightboxOverlay');
            var dialog = document.getElementById('csrLightboxDialog');
            var img = document.getElementById('csrLightboxImg');
            var btnClose = document.getElementById('csrLightboxClose');
            var btnPrev = document.getElementById('csrLightboxPrev');
            var btnNext = document.getElementById('csrLightboxNext');

            if (!overlay || !dialog || !img || !btnClose || !btnPrev || !btnNext) return;

            var lastActiveEl = null;
            var currentGroup = [];
            var currentIndex = -1;

            function setBodyScrollLocked(locked) {
                if (locked) {
                    document.documentElement.style.overflow = 'hidden';
                    document.body.style.overflow = 'hidden';
                } else {
                    document.documentElement.style.overflow = '';
                    document.body.style.overflow = '';
                }
            }

            function openAt(group, index) {
                if (!group || !group.length) return;
                currentGroup = group;
                currentIndex = Math.max(0, Math.min(index, group.length - 1));

                var href = group[currentIndex].getAttribute('href') || '';
                if (!href) return;

                lastActiveEl = document.activeElement;
                img.src = href;
                overlay.setAttribute('aria-hidden', 'false');
                overlay.setAttribute('data-has-nav', group.length > 1 ? 'true' : 'false');
                setBodyScrollLocked(true);
                dialog.focus();
            }

            function close() {
                overlay.setAttribute('aria-hidden', 'true');
                overlay.removeAttribute('data-has-nav');
                img.removeAttribute('src');
                setBodyScrollLocked(false);
                if (lastActiveEl && typeof lastActiveEl.focus === 'function') {
                    lastActiveEl.focus();
                }
                lastActiveEl = null;
                currentGroup = [];
                currentIndex = -1;
            }

            function go(delta) {
                if (!currentGroup.length) return;
                var nextIndex = (currentIndex + delta) % currentGroup.length;
